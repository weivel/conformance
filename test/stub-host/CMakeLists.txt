cmake_minimum_required(VERSION 3.15)
project(weivel_conformance_stub_host C)

# Configured on its own, the stub host builds against a standard checkout, as a language's suite builds it.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set(CMAKE_C_STANDARD 11)
    set(CMAKE_C_STANDARD_REQUIRED ON)
    set(CMAKE_C_EXTENSIONS OFF)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${PROJECT_BINARY_DIR}/bin)
    set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${PROJECT_BINARY_DIR}/bin)
    set(WEIVEL_STANDARD_SOURCE_DIR "" CACHE PATH "The standard checkout the stub host builds against")
    if(NOT EXISTS ${WEIVEL_STANDARD_SOURCE_DIR}/CMakeLists.txt)
        message(FATAL_ERROR "pass -DWEIVEL_STANDARD_SOURCE_DIR=<standard checkout> and "
            "-DWEIVEL_CORE_SOURCE_DIR=<core checkout>")
    endif()
    add_subdirectory(${WEIVEL_STANDARD_SOURCE_DIR} standard)
    function(conformance_strict_warnings target)
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
        endif()
    endfunction()
endif()

add_library(stub_host_objects OBJECT
    answers.c host.c lock.c record.c run_file.c scope.c values.c)
target_link_libraries(stub_host_objects PUBLIC weivel_standard)
target_include_directories(stub_host_objects PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(stub_host_objects PUBLIC STUB_HOST_BUILDING PRIVATE _CRT_SECURE_NO_WARNINGS)
set_target_properties(stub_host_objects PROPERTIES POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden)
conformance_strict_warnings(stub_host_objects)
if(NOT WIN32)
    find_package(Threads REQUIRED)
    target_link_libraries(stub_host_objects PUBLIC Threads::Threads m)
endif()

add_library(stub-host SHARED)
target_link_libraries(stub-host PRIVATE stub_host_objects)
